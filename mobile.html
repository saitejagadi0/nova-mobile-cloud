async function sendMessage() { const input = document.getElementById("messageInput"); const message = input.value.trim(); if (!message) return; input.value = ""; // Load existing memory let history = []; try { history = JSON.parse( localStorage.getItem("nova_chat_history") || "[]" ); } catch (e) { history = []; } // Show user message addMessage("user", message); try { const response = await fetch( NOVA_API + "/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: message, mode: currentMode, history: history }) } ); const data = await response.json(); if (!data.success) { addMessage( "assistant", "NOVA error: " + data.error ); return; } const answer = data.response; // Show NOVA response addMessage("assistant", answer); // IMPORTANT: // Save both sides of the conversation immediately. history.push({ role: "user", content: message }); history.push({ role: "assistant", content: answer }); // Keep last 20 messages history = history.slice(-20); localStorage.setItem( "nova_chat_history", JSON.stringify(history) ); speak(answer); } catch (error) { addMessage( "assistant", "Connection error: " + error.message ); } }
